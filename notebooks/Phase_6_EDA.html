late_rate = orders_df['IS_LATE'].mean() * 100

delivery_by_state = session.sql("""
    SELECT c.CUSTOMER_STATE,
           COUNT(*) AS TOTAL_ORDERS,
           ROUND(AVG(f.IS_LATE) * 100, 1) AS LATE_RATE_PCT
    FROM MARTS.FCT_ORDERS f
    JOIN STAGING.CUSTOMERS c ON f.CUSTOMER_ID = c.CUSTOMER_ID
    WHERE f.IS_LATE IS NOT NULL
    GROUP BY c.CUSTOMER_STATE
    ORDER BY LATE_RATE_PCT DESC
""").to_pandas()

import matplotlib.pyplot as plt
top10 = delivery_by_state.head(10)
plt.figure(figsize=(8,5))
plt.barh(top10['CUSTOMER_STATE'], top10['LATE_RATE_PCT'])
plt.xlabel('Late delivery rate (%)')
plt.title('Top 10 states by late delivery rate')
plt.gca().invert_yaxis()
plt.show()

seller_performance = session.sql("""
    SELECT oi.SELLER_ID,
           COUNT(DISTINCT oi.ORDER_ID) AS TOTAL_ORDERS,
           ROUND(AVG(f.IS_LATE) * 100, 1) AS LATE_RATE_PCT
    FROM MARTS.FCT_ORDER_ITEMS oi
    JOIN MARTS.FCT_ORDERS f ON oi.ORDER_ID = f.ORDER_ID
    WHERE f.IS_LATE IS NOT NULL
    GROUP BY oi.SELLER_ID
    HAVING COUNT(DISTINCT oi.ORDER_ID) >= 20
    ORDER BY LATE_RATE_PCT DESC
    LIMIT 10
""").to_pandas()
